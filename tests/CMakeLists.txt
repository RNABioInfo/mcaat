cmake_minimum_required(VERSION 3.16)
project(mcaat_protospacer_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(test_protospacer_extension test_protospacer_extension.cpp)
target_include_directories(test_protospacer_extension PRIVATE ../include)
add_test(NAME protospacer_extension COMMAND test_protospacer_extension)
add_executable(test_real_protospacer_fragments test_real_protospacer_fragments.cpp)
target_include_directories(test_real_protospacer_fragments PRIVATE ../include)
add_test(NAME real_protospacer_fragments COMMAND test_real_protospacer_fragments ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/protospacer-srr14017553)
option(PROTO_SANITIZERS "Enable address/undefined-behavior sanitizers" OFF)
if(PROTO_SANITIZERS)
    foreach(test_target test_protospacer_extension test_real_protospacer_fragments)
        target_compile_options(${test_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${test_target} PRIVATE -fsanitize=address,undefined)
    endforeach()
endif()
